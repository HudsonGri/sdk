#!/usr/bin/env bash
set -euo pipefail
source .bb-plugins-target-guard.sh
export CARGO_INCREMENTAL=0 RUSTC_WRAPPER=/home/var/bin/sccache-rustc RUSTC_WORKSPACE_WRAPPER=''
p=.bb-plugins-lifecycle-final
mkdir -p "$p/formatted"
step=format-final-lifecycle-contracts
trap 'rc=$?; if ((rc)); then echo "step=$step failed($rc); recovery: pwsh -NoProfile -File Q:/lanes/bs.ps1 --env GIT_CONFIG_COUNT=1 --env GIT_CONFIG_KEY_0=core.excludesFile --env GIT_CONFIG_VALUE_0=/mnt/d/lane-scratch/plugins/build-source-excludes.txt bash .bb-plugins-lifecycle-final.sh" >&2; fi' EXIT
rustfmt --edition 2024 --config skip_children=true rust/crates/harness-core/src/task_volume/child_queries.rs rust/crates/harness/src/local/host_owner/sessions.rs rust/crates/harness/src/local/host_owner/tests.rs rust/crates/filesystem/src/native_mount/tests/fence.rs
cp --parents rust/crates/harness-core/src/task_volume/child_queries.rs rust/crates/harness/src/local/host_owner/sessions.rs rust/crates/harness/src/local/host_owner/tests.rs rust/crates/filesystem/src/native_mount/tests/fence.rs "$p/formatted"
step=final-owner-and-mailbox-build
cargo test --locked -p acyclic-harness -p acyclic-harness-core -p acyclic-fs --all-features --lib --no-run --message-format=json > "$p/cargo.jsonl" 2> "$p/build.log"
step=final-owner-and-mailbox-contracts
node .bb-plugins-run-suites.mjs "$p/cargo.jsonl" 'acyclic-harness::acyclic_harness=local::host_owner::' 'acyclic-harness-core::acyclic_harness_core=task_volume::child_queries::tests::' 'acyclic-fs::acyclic_fs=native_mount::tests::fence::' > "$p/tests.log" 2>&1
cp .bb-plugins-latest-suites.json "$p/results.json"
step=final-normal-and-scale-shipping
bash .bb-plugins-consolidated-shipping.sh > "$p/shipping.log" 2>&1
step=final-attested-clippy
set +e
cargo clippy --locked --keep-going -p acyclic-conformance -p acyclic-fs-napi -p acyclic-harness-search -p acyclic-objects -p acyclic-stream -p acyclic-native-runtime -p acyclic-plugin-identity -p acyclic-harness-core -p acyclic-harness -p acyclic-plugin -p graphcoder -p acyclic-fs -p acyclic-harness-fuzz -p acyclic-fs-fuzz --all-targets --all-features --message-format=short -- --cap-lints warn > "$p/clippy.out" 2>&1
clippy_rc=$?
set -e
printf 'GATE_CLIPPY rc=%s\n' "$clippy_rc" > "$p/clippy-qualified.log"
python3 .bb-plugins-canonical-gate-clippy.py emit "$p/clippy.out" "$clippy_rc" >> "$p/clippy-qualified.log"
python3 .bb-plugins-canonical-gate-clippy.py compare "$p/clippy-qualified.log" .bb-plugins-consolidated-main-clippy/qualified.log > "$p/clippy-verdict.log"
step=actual-previous-projection-negative
bash .bb-plugins-previous-answer-negative.sh > "$p/negative.log" 2>&1

#!/usr/bin/env bash
set -euo pipefail
source .bb-plugins-target-guard.sh
export CARGO_INCREMENTAL=0 RUSTC_WRAPPER=/home/var/bin/sccache-rustc RUSTC_WORKSPACE_WRAPPER=''
step=exact-main-clippy-control
trap 'rc=$?; if ((rc)); then echo "step=$step failed($rc); recovery: pwsh -NoProfile -File Q:/lanes/bs.ps1 --env GIT_CONFIG_COUNT=1 --env GIT_CONFIG_KEY_0=core.excludesFile --env GIT_CONFIG_VALUE_0=/mnt/d/lane-scratch/plugins/build-source-excludes.txt bash .bb-plugins-consolidated-main-clippy.sh" >&2; fi' EXIT
proof="$PWD/.bb-plugins-consolidated-main-clippy"
mkdir -p "$proof"
main_source=$(mktemp -d "$CARGO_TARGET_DIR/unchanged-main.XXXXXX")
git archive b7333953cbd5d3b31adc05cf2f0851f049f6c16a | tar -x -C "$main_source"
cd "$main_source"
cargo clippy --locked --keep-going -p acyclic-conformance -p acyclic-fs-napi -p acyclic-harness-search -p acyclic-objects -p acyclic-stream -p acyclic-native-runtime -p acyclic-plugin-identity -p acyclic-harness-core -p acyclic-harness -p acyclic-plugin -p graphcoder -p acyclic-fs -p acyclic-harness-fuzz -p acyclic-fs-fuzz --all-targets --all-features --message-format=short -- --cap-lints warn > "$proof/clippy.out" 2>&1
printf 'GATE_CLIPPY rc=0\n' > "$proof/qualified.log"
python3 /work/src/.bb-plugins-canonical-gate-clippy.py emit "$proof/clippy.out" 0 >> "$proof/qualified.log"
printf '%s\n' b7333953cbd5d3b31adc05cf2f0851f049f6c16a > "$proof/source"
